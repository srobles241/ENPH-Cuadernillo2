CREATE OR REPLACE PROCEDURE enph_registrar_aud(
 p_tabla IN VARCHAR2,p_registro IN VARCHAR2,p_accion IN VARCHAR2,p_contexto IN VARCHAR2
) IS
 n NUMBER;
BEGIN
 IF p_contexto IS NULL OR SUBSTR(p_contexto,1,8)<>'ENPHADM|' THEN
   RAISE_APPLICATION_ERROR(-20145,'Falta el contexto de auditoria administrativa.');
 END IF;
 -- Mientras siga activo el disparador anterior de la tabla, el registra el cambio.
 SELECT COUNT(*) INTO n FROM user_triggers
 WHERE trigger_name='ENPHA_AUD_'||p_tabla AND status='ENABLED';
 IF n=0 THEN
   INSERT INTO enph_aud_admin(id_aud,tabla_objeto,id_registro,accion,contexto,marca)
   VALUES(enph_seq_aud_admin.NEXTVAL,p_tabla,p_registro,p_accion,p_contexto,'Editado por el administrador');
 END IF;
END;

CREATE OR REPLACE TRIGGER enph_val_visita_ciclo
 BEFORE INSERT OR UPDATE OF id_hogar,fecha_visita,numero_dia_ciclo ON visita
 FOR EACH ROW
DECLARE
 v_inicio DATE;
 v_dia NUMBER;
BEGIN
 SELECT fecha_inicio_ciclo INTO v_inicio FROM hogar WHERE id_hogar=:NEW.id_hogar;
 v_dia:=TRUNC(:NEW.fecha_visita)-TRUNC(v_inicio)+1;
 IF v_dia IS NULL OR v_dia NOT BETWEEN 1 AND 14 THEN
   RAISE_APPLICATION_ERROR(-20146,'La fecha de la visita esta fuera del ciclo de 14 dias del hogar.');
 END IF;
 IF :NEW.numero_dia_ciclo IS NULL OR :NEW.numero_dia_ciclo<>v_dia THEN
   RAISE_APPLICATION_ERROR(-20147,'El dia de la visita no coincide con la fecha y el ciclo del hogar.');
 END IF;
EXCEPTION
 WHEN NO_DATA_FOUND THEN
   RAISE_APPLICATION_ERROR(-20148,'El hogar indicado para la visita no existe.');
END;
/
SHOW ERRORS TRIGGER enph_val_visita_ciclo

CREATE OR REPLACE TRIGGER enpha_aud_hogar
AFTER INSERT OR UPDATE OR DELETE ON HOGAR FOR EACH ROW
DECLARE
 v_contexto VARCHAR2(64):=SYS_CONTEXT('USERENV','CLIENT_IDENTIFIER');
 v_accion VARCHAR2(20);
 v_registro VARCHAR2(110);
BEGIN
 IF SUBSTR(v_contexto,1,8)='ENPHADM|' THEN
  IF INSERTING THEN v_accion:='CREAR';v_registro:=TO_CHAR(:NEW.ID_HOGAR);
  ELSIF DELETING THEN v_accion:='ELIMINAR';v_registro:=TO_CHAR(:OLD.ID_HOGAR);
  ELSE v_accion:='EDITAR';v_registro:=TO_CHAR(:NEW.ID_HOGAR);
  
  END IF;
  INSERT INTO enph_aud_admin(id_aud,tabla_objeto,id_registro,accion,contexto,marca)
  VALUES(enph_seq_aud_admin.NEXTVAL,'HOGAR',v_registro,v_accion,v_contexto,'Editado por el administrador');
 END IF;
END;
