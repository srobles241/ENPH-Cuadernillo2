-- UDI. Instalar despues de los TXT 10 y 11, con Java cerrado.
WHENEVER SQLERROR EXIT SQL.SQLCODE
DECLARE
 n NUMBER;
BEGIN
 IF USER <> 'PROYECTOINTEGRADOR' THEN
   RAISE_APPLICATION_ERROR(-20134,'Conectate como PROYECTOINTEGRADOR en la UDI.');
 END IF;
 SELECT COUNT(*) INTO n FROM user_triggers WHERE status='ENABLED' AND trigger_name IN
 ('ENPH_VAL_VISITA_CICLO','ENPH_VAL_GASTO_VISITA','ENPHA_AUD_HOGAR','ENPHA_AUD_VISITA');
 IF n<>4 THEN RAISE_APPLICATION_ERROR(-20157,'Instala primero los TXT 10 y 11.'); END IF;
 SELECT COUNT(*) INTO n FROM user_triggers WHERE trigger_name NOT IN
 ('ENPH_VAL_VISITA_CICLO','ENPH_VAL_GASTO_VISITA','ENPHA_AUD_HOGAR','ENPHA_AUD_VISITA',
  'ENPHA_AUD_CASO','ENPH_VAL_APROBAR_CASO');
 IF n<>0 THEN RAISE_APPLICATION_ERROR(-20152,'El inventario de disparadores no coincide.'); END IF;
 SELECT COUNT(*) INTO n FROM user_objects WHERE object_name='ENPH_REGISTRAR_AUD'
 AND object_type='PROCEDURE' AND status='VALID';
 IF n<>1 THEN RAISE_APPLICATION_ERROR(-20153,'Falta el procedimiento de auditoria.'); END IF;
END;
/
CREATE OR REPLACE TRIGGER enph_val_aprobar_caso
 BEFORE INSERT OR UPDATE OF estado_caso,id_hogar ON caso
 FOR EACH ROW
DECLARE
 v_dias NUMBER;
 v_pendientes NUMBER;
BEGIN
 IF :NEW.estado_caso='Aprobado' THEN
   IF INSERTING OR :OLD.estado_caso IS NULL OR :OLD.estado_caso<>'Aprobado'
      OR :OLD.id_hogar<>:NEW.id_hogar THEN
     SELECT COUNT(DISTINCT CASE WHEN numero_dia_ciclo BETWEEN 1 AND 14 THEN numero_dia_ciclo END),
            COUNT(CASE WHEN estado_visita='Revisar' THEN 1 END)
       INTO v_dias,v_pendientes
       FROM visita WHERE id_hogar=:NEW.id_hogar;
     IF v_dias<14 THEN
       RAISE_APPLICATION_ERROR(-20158,'No se puede aprobar el caso. Faltan '||(14-v_dias)||' dias de visita.');
     END IF;
     IF v_pendientes>0 THEN
       RAISE_APPLICATION_ERROR(-20159,'No se puede aprobar el caso. Hay '||v_pendientes||' visitas por revisar.');
     END IF;
   END IF;
 END IF;
END;
/
SHOW ERRORS TRIGGER enph_val_aprobar_caso
DECLARE
 n NUMBER;
BEGIN
 SELECT COUNT(*) INTO n FROM user_objects WHERE object_name='ENPH_VAL_APROBAR_CASO'
 AND object_type='TRIGGER' AND status='VALID';
 IF n<>1 THEN RAISE_APPLICATION_ERROR(-20160,'Revisa la compilacion del disparador nuevo.'); END IF;
END;
