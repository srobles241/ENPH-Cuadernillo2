DECLARE
 n NUMBER;
BEGIN
 IF USER <> 'PROYECTOINTEGRADOR' THEN
   RAISE_APPLICATION_ERROR(-20134,'Conectate como PROYECTOINTEGRADOR en la UDI.');
 END IF;
 SELECT COUNT(*) INTO n FROM session_privs
 WHERE privilege IN ('CREATE USER','CREATE ROLE','CREATE PROCEDURE');
 IF n<>3 THEN
   RAISE_APPLICATION_ERROR(-20136,'Faltan CREATE USER, CREATE ROLE o CREATE PROCEDURE en la sesion.');
 END IF;
 SELECT COUNT(*) INTO n FROM user_objects
 WHERE status='VALID' AND (
   (object_type='PROCEDURE' AND object_name='ENPH_PROVISIONAR_ESQUEMA')
   OR (object_type='VIEW' AND object_name IN ('ENPH_U_V_CATALOGO','ENPH_U_V_RECORRIDO','ENPH_U_V_REVISION'))
 );
 IF n<>4 THEN
   RAISE_APPLICATION_ERROR(-20138,'Primero verifica la instalacion de ENPH-UDI-06-esquemas.txt.');
 END IF;
 SELECT COUNT(*) INTO n FROM user_tab_columns
 WHERE table_name='ENPH_USUARIO' AND column_name='USUARIO_ORACLE';
 IF n<>1 THEN
   RAISE_APPLICATION_ERROR(-20139,'Falta ENPH_USUARIO.USUARIO_ORACLE.');
 END IF;
END;
/

CREATE ROLE udi_enph_encuestador;
CREATE ROLE udi_enph_recuentista;
CREATE ROLE udi_enph_supervisor;

GRANT SELECT ON enph_u_v_catalogo TO udi_enph_encuestador;
GRANT SELECT ON enph_u_v_recorrido TO udi_enph_recuentista;
GRANT SELECT ON enph_u_v_revision TO udi_enph_supervisor;

-- Conserva los roles anteriores y agrega el distintivo a los esquemas existentes.
DECLARE
 v_rol VARCHAR2(30);
BEGIN
 FOR u IN (
   SELECT e.usuario_oracle, e.rol
   FROM enph_usuario e JOIN all_users a ON a.username=e.usuario_oracle
   WHERE e.rol IN ('Encuestador','Recuentista','Supervisor')
     AND REGEXP_LIKE(e.usuario_oracle,'^ENPH_U_[A-Z0-9_]{1,23}$')
 ) LOOP
   CASE u.rol
     WHEN 'Encuestador' THEN v_rol:='UDI_ENPH_ENCUESTADOR';
     WHEN 'Recuentista' THEN v_rol:='UDI_ENPH_RECUENTISTA';
     WHEN 'Supervisor' THEN v_rol:='UDI_ENPH_SUPERVISOR';
   END CASE;
   EXECUTE IMMEDIATE 'GRANT '||v_rol||' TO '||u.usuario_oracle;
 END LOOP;
END;
/

CREATE OR REPLACE PROCEDURE enph_provisionar_esquema(
 p_usuario IN VARCHAR2,p_rol IN VARCHAR2,p_clave IN VARCHAR2
) AUTHID CURRENT_USER IS
 v_rol VARCHAR2(30);
 v_creado BOOLEAN:=FALSE;
BEGIN
 IF SYS_CONTEXT('USERENV','SESSION_USER') <> 'PROYECTOINTEGRADOR' THEN
   RAISE_APPLICATION_ERROR(-20135,'Solo el propietario de ENPH en UDI puede ejecutar el alta.');
 END IF;
 IF p_usuario IS NULL OR NOT REGEXP_LIKE(p_usuario,'^ENPH_U_[A-Z0-9_]{1,23}$')
    OR p_clave IS NULL OR NOT REGEXP_LIKE(p_clave,'^[A-Za-z][A-Za-z0-9]{19,29}$') THEN
   RAISE_APPLICATION_ERROR(-20130,'Nombre o clave de provision invalida.');
 END IF;
 CASE p_rol
   WHEN 'Encuestador' THEN v_rol:='UDI_ENPH_ENCUESTADOR';
   WHEN 'Recuentista' THEN v_rol:='UDI_ENPH_RECUENTISTA';
   WHEN 'Supervisor' THEN v_rol:='UDI_ENPH_SUPERVISOR';
   ELSE RAISE_APPLICATION_ERROR(-20131,'Rol operativo invalido.');
 END CASE;
 BEGIN
   EXECUTE IMMEDIATE 'CREATE USER '||p_usuario||' IDENTIFIED BY "'||p_clave||'" ACCOUNT LOCK';
   v_creado:=TRUE;
   EXECUTE IMMEDIATE 'GRANT '||v_rol||' TO '||p_usuario;
 EXCEPTION WHEN OTHERS THEN
   IF v_creado THEN
     RAISE_APPLICATION_ERROR(-20132,'Esquema bloqueado creado sin completar rol: '||p_usuario,TRUE);
   END IF;
   RAISE;
 END;
END;
/
SHOW ERRORS PROCEDURE enph_provisionar_esquema
DECLARE n NUMBER;
BEGIN
 SELECT COUNT(*) INTO n FROM user_objects
 WHERE object_name='ENPH_PROVISIONAR_ESQUEMA' AND object_type='PROCEDURE' AND status='VALID';
 IF n<>1 THEN RAISE_APPLICATION_ERROR(-20133,'Provisionador invalido. Revisa SHOW ERRORS.'); END IF;
END;
/
