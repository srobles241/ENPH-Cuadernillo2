CREATE OR REPLACE TRIGGER enph_val_gasto_visita
 BEFORE INSERT OR UPDATE ON gasto
 FOR EACH ROW
DECLARE
 PROCEDURE validar_visita(p_visita NUMBER) IS
   v_estado visita.estado_visita%TYPE;
 BEGIN
   SELECT estado_visita INTO v_estado FROM visita WHERE id_visita=p_visita FOR UPDATE;
   IF v_estado='Completa' THEN
     RAISE_APPLICATION_ERROR(-20154,'No se puede registrar o corregir un gasto en una visita completa.');
   END IF;
 EXCEPTION
   WHEN NO_DATA_FOUND THEN
     RAISE_APPLICATION_ERROR(-20155,'La visita indicada para el gasto no existe.');
 END;
BEGIN
 IF UPDATING AND :OLD.id_visita<>:NEW.id_visita THEN
   validar_visita(LEAST(:OLD.id_visita,:NEW.id_visita));
   validar_visita(GREATEST(:OLD.id_visita,:NEW.id_visita));
 ELSE
   validar_visita(:NEW.id_visita);
 END IF;
END;
/

END;
