CREATE OR REPLACE FUNCTION enph3_dia_ciclo(p_inicio DATE, p_fecha DATE)
RETURN NUMBER IS
  v_dia NUMBER;
BEGIN
  IF p_inicio IS NULL OR p_fecha IS NULL THEN
    RAISE_APPLICATION_ERROR(-20031,'La fecha del ciclo es obligatoria.');
  END IF;
  v_dia := TRUNC(p_fecha) - TRUNC(p_inicio) + 1;
  IF v_dia NOT BETWEEN 1 AND 14 THEN
    RAISE_APPLICATION_ERROR(-20032,'La fecha esta fuera del ciclo de 14 dias.');
  END IF;
  RETURN v_dia;
END;
/

CREATE OR REPLACE FUNCTION enph3_total_hogar(p_hogar NUMBER)
RETURN NUMBER IS
  v_total NUMBER;
BEGIN
  SELECT NVL(SUM(g.valor_pagado),0) INTO v_total
  FROM gasto g JOIN visita v ON v.id_visita=g.id_visita
  WHERE v.id_hogar=p_hogar AND v.numero_dia_ciclo BETWEEN 1 AND 14;
  RETURN v_total;
END;
/
